// CSV 업로드 파싱 로직 (EUC-KR / CP949 / UTF-8 완전 자동 감지) function handleCSVUpload(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(e) { const buffer = e.target.result; let text = ""; try { // 1. Strict UTF-8 디코딩 시도 (잘못된 인코딩이면 에러 발생) const utf8Decoder = new TextDecoder('utf-8', { fatal: true }); text = utf8Decoder.decode(buffer); } catch (err) { // 2. UTF-8 디코딩 실패 시 EUC-KR / CP949(한글 윈도우 엑셀)로 디코딩 try { const eucKrDecoder = new TextDecoder('euc-kr'); text = eucKrDecoder.decode(buffer); } catch (eucErr) { const ansiDecoder = new TextDecoder('iso-8859-1'); text = ansiDecoder.decode(buffer); } } parseCSVData(text); }; reader.readAsArrayBuffer(file); }
