function renderCard() { if (currentDeck.length === 0) { articleText.textContent = "조건에 해당하는 카드가 없습니다."; reasonText.textContent = ""; progressText.textContent = "0 / 0"; progressBar.style.width = '0%'; btnAnswerO.disabled = true; btnAnswerX.disabled = true; return; } btnAnswerO.disabled = false; btnAnswerX.disabled = false; // 1. 카드가 뒤집혀 있다면 먼저 '빈 내용'으로 보이지 않게 처리 후 앞면으로 복귀 if (isFlipped) { unflipCardSilently(); } // 2. 앞면으로 완전히 돌아온 후 데이터를 세팅 (답 스포일러 완벽 방지) const currentCard = currentDeck[currentIndex]; articleText.textContent = currentCard.article || "조문 내용이 없습니다."; reasonText.textContent = currentCard.reason || "상세 사유가 없습니다."; const rawAns = (currentCard.answer || "O").trim().toUpperCase(); correctAnswerDisplay.textContent = rawAns; if (rawAns === 'O') { correctAnswerDisplay.className = "text-2xl font-black text-emerald-600"; } else { correctAnswerDisplay.className = "text-2xl font-black text-rose-600"; } const key = getCardKey(currentCard); const pastAnswer = userAnswers[key]; if (pastAnswer) { showResultBadge(pastAnswer.isCorrect, pastAnswer.userAns); } else { resultBadge.className = "hidden"; } btnPrev.disabled = currentIndex === 0; btnNext.disabled = currentIndex === currentDeck.length - 1; const total = currentDeck.length; const currentNum = currentIndex + 1; progressText.textContent = `${currentNum} / ${total}`; progressBar.style.width = `${(currentNum / total) * 100}%`; } function unflipCardSilently() { isFlipped = false; // 카드가 돌아갈 때 뒷면 텍스트를 순간적으로 숨겨 스포일러 방지 correctAnswerDisplay.textContent = ""; reasonText.textContent = ""; // 트랜지션을 끄고 즉시 앞면 상태로 리셋 flashcard.style.transition = 'none'; flashcard.classList.remove('rotate-y-180'); // DOM 리플로우 강제 후 트랜지션 원복 void flashcard.offsetHeight; flashcard.style.transition = ''; }
